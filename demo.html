import okhttp3.*;
import java.io.File;
import java.io.IOException;

public class GenImage {
    public static void main(String[] args) throws IOException {
        OkHttpClient client = new OkHttpClient();

        MultipartBody.Builder bodyBuilder = new MultipartBody.Builder().setType(MultipartBody.FORM);
        bodyBuilder.addFormDataPart("prompt", "Create a dog eat pizza");
        bodyBuilder.addFormDataPart("model", "imagen-flash");

        Request request = new Request.Builder()
                .url("https://api.geminigen.ai/uapi/v1/generate_image")
                .addHeader("x-api-key", "your_api_key")
                .addHeader("Accept", "application/json")
                .post(bodyBuilder.build())
                .build();

        try (Response response = client.newCall(request).execute()) {
            System.out.println(response.body().string());
        }
    }
}

import java.io.*;
import java.net.HttpURLConnection;
import java.net.URL;

public class Main {
    public static void main(String[] args) {
        try {
            String boundary = "----WebKitFormBoundary7MA4YWxkTrZu0gW";
            URL url = new URL("https://api.geminigen.ai/uapi/v1/video-gen/veo");
            HttpURLConnection conn = (HttpURLConnection) url.openConnection();
            conn.setRequestMethod("POST");
            conn.setDoOutput(true);
            conn.setRequestProperty("Accept", "application/json");
            conn.setRequestProperty("x-api-key", "your_api_key");
            conn.setRequestProperty("Content-Type", "multipart/form-data; boundary=" + boundary);

            OutputStream os = conn.getOutputStream();
            PrintWriter writer = new PrintWriter(new OutputStreamWriter(os, "UTF-8"), true);

            // prompt
            writer.append("--" + boundary).append("\r\n");
            writer.append("Content-Disposition: form-data; name=\"prompt\"\r\n\r\n");
            writer.append("Create a cat small").append("\r\n");

            // model
            writer.append("--" + boundary).append("\r\n");
            writer.append("Content-Disposition: form-data; name=\"model\"\r\n\r\n");
            writer.append("veo-2").append("\r\n");

            // enhance_prompt
            writer.append("--" + boundary).append("\r\n");
            writer.append("Content-Disposition: form-data; name=\"enhance_prompt\"\r\n\r\n");
            writer.append("true").append("\r\n");

            // other params...

            writer.append("--" + boundary + "--").append("\r\n");
            writer.flush();
            writer.close();

            BufferedReader br = new BufferedReader(new InputStreamReader(conn.getInputStream(), "UTF-8"));
            String line;
            StringBuilder response = new StringBuilder();
            while ((line = br.readLine()) != null) {
                response.append(line).append("\n");
            }
            br.close();
            System.out.println(response.toString());

        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
